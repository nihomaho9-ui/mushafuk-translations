تَرْهَقُهَا قَتَرَةٌ ﴿41﴾

أُو۟لَـٰٓئِكَ هُمُ ٱلْكَفَرَةُ ٱلْفَجَرَةُ ﴿42﴾

بِسْمِ ٱللَّهِ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ إِذَا ٱلشَّمْسُ كُوِّرَتْ ﴿1﴾

وَإِذَا ٱلنُّجُومُ ٱنكَدَرَتْ ﴿2﴾

وَإِذَا ٱلْجِبَالُ سُيِّرَتْ ﴿3﴾

وَإِذَا ٱلْعِشَارُ عُطِّلَتْ ﴿4﴾

وَإِذَا ٱلْوُحُوشُ حُشِرَتْ ﴿5﴾

وَإِذَا ٱلْبِحَارُ سُجِّرَتْ ﴿6﴾

وَإِذَا ٱلنُّفُوسُ زُوِّجَتْ ﴿7﴾

وَإِذَا ٱلْمَوْءُۥدَةُ سُئِلَتْ ﴿8﴾

بِأَىِّ ذَنۢبٍ قُتِلَتْ ﴿9﴾

وَإِذَا ٱلصُّحُفُ نُشِرَتْ ﴿10﴾

وَإِذَا ٱلسَّمَآءُ كُشِطَتْ ﴿11﴾

وَإِذَا ٱلْجَحِيمُ سُعِّرَتْ ﴿12﴾

وَإِذَا ٱلْجَنَّةُ أُزْلِفَتْ ﴿13﴾

عَلِمَتْ نَفْسٌ مَّآ أَحْضَرَتْ ﴿14﴾

فَلَآ أُقْسِمُ بِٱلْخُنَّسِ ﴿15﴾

ٱلْجَوَارِ ٱلْكُنَّسِ ﴿16﴾

وَٱلَّيْلِ إِذَا عَسْعَسَ ﴿17﴾

وَٱلصُّبْحِ إِذَا تَنَفَّسَ ﴿18﴾

إِنَّهُۥ لَقَوْلُ رَسُولٍ كَرِيمٍ ﴿19﴾

ذِى قُوَّةٍ عِندَ ذِى ٱلْعَرْشِ مَكِينٍ ﴿20﴾

مُّطَاعٍ ثَمَّ أَمِينٍ ﴿21﴾

وَمَا صَاحِبُكُم بِمَجْنُونٍ ﴿22﴾

وَلَقَدْ رَءَاهُ بِٱلْأُفُقِ ٱلْمُبِينِ ﴿23﴾

وَمَا هُوَ عَلَى ٱلْغَيْبِ بِضَنِينٍ ﴿24﴾

وَمَا هُوَ بِقَوْلِ شَيْطَـٰنٍ رَّجِيمٍ ﴿25﴾

فَأَيْنَ تَذْهَبُونَ ﴿26﴾

إِنْ هُوَ إِلَّا ذِكْرٌ لِّلْعَـٰلَمِينَ ﴿27﴾

لِمَن شَآءَ مِنكُمْ أَن يَسْتَقِيمَ ﴿28﴾

وَمَا تَشَآءُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ رَبُّ ٱلْعَـٰلَمِينَ ﴿29﴾

Ali Özek ve diğerleri

80:41 hüzünden kapkara kesilmiştir.

80:42 İşte bunlar kâfirlerdir, günahkârlardır.

81:1 Güneş katlanıp dürüldüğünde,

81:2 Yıldızlar (kararıp) döküldüğünde,

81:3 Dağlar (sallanıp) yürütüldüğünde,

81:4 Gebe develer salıverildiğinde,

81:5 Vahşî hayvanlar toplanıp bir araya getirildiğinde,

81:6 Denizler kaynatıldığında,

81:7 Ruhlar (bedenlerle) birleştirildiğinde,

81:8 Diri diri toprağa gömülen kıza, sorulduğunda,

81:9 hangi günah sebebiyle öldürüldüğü,

81:10 (Amellerin yazılı olduğu) defterler açıldığında,

81:11 Gökyüzü sıyrılıp alındığında,

81:12 cehennem tutuşturulduğunda

81:13 ve cennet yaklaştırıldığında,

81:14 Kişi neler getirdiğini öğrenmiş olacaktır.

81:15 Hayır! (Gündüz) sinen,

81:16 akıp giden, etrafı aydınlatan yıldızlara andolsun,

81:17 Kararmaya yüz tuttuğunda geceye andolsun,

81:18 Ağarmaya başladığında sabaha andolsun ki,

81:19 itibarlı bir elçinin (Cebrail'in) getirdiği sözdür.

81:20 O (Kur'an), şüphesiz değerli, güçlü ve Arş'ın sahibi (Allah'ın) katında

81:21 O orada sayılan, güvenilen (elçi)'dir.

81:22 Arkadaşınız (Muhammed) de mecnun değildir.

81:23 Andolsun ki, onu (Cebrail'i) apaçık ufukta görmüştür.

81:24 O, gaybın bilgilerini (sizden) esirgemez.

81:25 O lanetlenmiş şeytanın sözü de değildir.

81:26 Hal böyle iken nereye gidiyorsunuz?

81:27 O, herkes için, bir öğüttür.

81:28 Sizden doğru yolda gitmek isteyenler için.

81:29 Âlemlerin Rabbi Allah dilemedikçe siz dileyemezsiniz.
